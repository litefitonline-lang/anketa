function dnEsc_(s) { return String(s == null ? '' : s).replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;'); }

function dnSplit_(text, max) {
  text = String(text || '');
  if (text.length <= max) return [text];
  const out = [];
  let cur = '';
  const push = function (piece, sep) {
    if (!cur) { cur = piece; return; }
    if ((cur + sep + piece).length <= max) cur += sep + piece; else { out.push(cur); cur = piece; }
  };
  text.split(/\n\n/).forEach(function (par) {
    if (par.length <= max) { push(par, '\n\n'); return; }
    par.split(/\n/).forEach(function (line) {
      while (line.length > max) { push(line.slice(0, max), '\n'); line = line.slice(max); }
      push(line, '\n');
    });
  });
  if (cur) out.push(cur);
  return out;
}

function tg_(method, params, multipart) {
  const url = 'https://api.telegram.org/bot' + dnGet_('TG_TOKEN') + '/' + method;
  for (let i = 0; i < 3; i++) {
    let opt;
    if (multipart) {
      const p = {};
      Object.keys(params).forEach(function (k) {
        const v = params[k];
        if (v === undefined || v === null) return;
        p[k] = v.getBytes ? v : (typeof v === 'object' ? JSON.stringify(v) : String(v));
      });
      opt = { method: 'post', payload: p, muteHttpExceptions: true };
    } else {
      opt = { method: 'post', contentType: 'application/json', payload: JSON.stringify(params), muteHttpExceptions: true };
    }
    let j;
    try { j = JSON.parse(UrlFetchApp.fetch(url, opt).getContentText()); }
    catch (e) { j = { ok: false, description: e.message }; }
    if (j.ok) return j;
    if (j.error_code === 429 && i < 2) { Utilities.sleep(((j.parameters && j.parameters.retry_after) || 2) * 1000); continue; }
    return j;
  }
  return { ok: false, description: 'нет ответа' };
}
function tgErr_(j) { return (j && (j.description || j.error_code)) || 'ошибка'; }
function dnSay_(text, extra) {
  const chat = dnGet_('OWNER_ID');
  if (!chat) return null;
  const p = { chat_id: chat, text: text, parse_mode: 'HTML', disable_web_page_preview: true };
  Object.keys(extra || {}).forEach(function (k) { p[k] = extra[k]; });
  return tg_('sendMessage', p);
}
function dnFileBlob_(fileId, name) {
  const f = tg_('getFile', { file_id: fileId });
  if (!f.ok) throw new Error('Telegram не отдал файл: ' + tgErr_(f));
  const r = UrlFetchApp.fetch('https://api.telegram.org/file/bot' + dnGet_('TG_TOKEN') + '/' + f.result.file_path, { muteHttpExceptions: true });
  if (r.getResponseCode() !== 200) throw new Error('файл Telegram не скачался: ' + r.getResponseCode());
  return r.getBlob().setName(name);
}

let DN_SH = null;
function dnSheet_() {
  if (DN_SH) return DN_SH;
  const id = dnGet_('SS_ID');
  if (!id) throw new Error('Таблица не создана – запустите dnSetup');
  DN_SH = SpreadsheetApp.openById(id).getSheetByName(DN.SHEET);
  return DN_SH;
}

function dnNewId_() { return 'd' + dnFmt_(new Date(), 'yyMMddHHmmss') + dnRand_(4); }

/** Новая запись. tag: '' | 'idea' | 'personal'; type: 'текст' | 'голос'; audio – ID файла на Диске. */
function dnAdd_(text, tag, type, audio) {
  text = String(text || '').trim();
  if (!text && !audio) throw new Error('Пустая запись');
  const now = new Date(), id = dnNewId_();
  const row = [dnFmt_(now, 'dd.MM.yyyy'), dnFmt_(now, 'HH:mm'), DN.TAGS[tag] || '', type || 'текст', dnCell_(text),
    audio ? 'https://drive.google.com/file/d/' + audio + '/view' : '', id, dnFmt_(now, 'yyyy-MM-dd')];
  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try { dnSheet_().appendRow(row); } finally { lock.releaseLock(); }
  return { id: id, time: row[1], date: row[0], t: now.getTime() };
}

function dnFindRow_(id) {
  if (!id) return 0;
  const sh = dnSheet_(), last = sh.getLastRow();
  if (last < 2) return 0;
  const c = sh.getRange(2, 7, last - 1, 1).createTextFinder(String(id)).matchEntireCell(true).findNext();
  return c ? c.getRow() : 0;
}
function dnSetTag_(id, tag) {
  const r = dnFindRow_(id);
  if (!r) return false;
  dnSheet_().getRange(r, 3).setValue(DN.TAGS[tag] || '');
  return true;
}
/** Удалить запись, если ей меньше UNDO_MIN минут (ID содержит время создания). */
function dnUndo_(id) {
  const m = /^d(\d{12})/.exec(String(id || ''));
  if (!m) return 'нет такой записи';
  const s = m[1], made = Utilities.parseDate('20' + s, DN.TZ, 'yyyyMMddHHmmss').getTime();
  if (Date.now() - made > DN.UNDO_MIN * 60000) return 'отменить можно только в первые ' + DN.UNDO_MIN + ' минут – удалите строку в таблице';
  const r = dnFindRow_(id);
  if (!r) return 'запись уже удалена';
  const audio = String(dnSheet_().getRange(r, 6).getValue() || '');
  dnSheet_().deleteRow(r);
  const fm = /\/d\/([^/]+)/.exec(audio);
  if (fm) { try { DriveApp.getFileById(fm[1]).setTrashed(true); } catch (e) {} }
  return '';
}

function dnTodayCount_() {
  const sh = dnSheet_(), last = sh.getLastRow();
  if (last < 2) return 0;
  const today = dnFmt_(new Date(), 'yyyy-MM-dd'), from = Math.max(2, last - 300);
  return sh.getRange(from, 8, last - from + 1, 1).getDisplayValues().filter(function (r) { return r[0] === today; }).length;
}

/** Запустить из редактора после первого развёртывания (и после смены токена). Итог – в журнале выполнения. */
function dnSetup() {
  const L = ['Дневник ' + DN.VERSION];
  if (!dnGet_('TG_TOKEN')) throw new Error('Впишите TG_TOKEN в свойства скрипта');
  if (!dnGet_('WH_KEY')) dnSet_('WH_KEY', dnRand_(32));
  if (!dnGet_('SETUP_CODE')) dnSet_('SETUP_CODE', dnRand_(6).toUpperCase());

  // таблица
  let ss = null;
  if (dnGet_('SS_ID')) { try { ss = SpreadsheetApp.openById(dnGet_('SS_ID')); } catch (e) { ss = null; } }
  if (!ss) { ss = SpreadsheetApp.create(DN.SS_TITLE); dnSet_('SS_ID', ss.getId()); }
  ss.setSpreadsheetTimeZone(DN.TZ);
  let sh = ss.getSheetByName(DN.SHEET);
  if (!sh) { sh = ss.getSheets().length === 1 && ss.getSheets()[0].getLastRow() === 0 ? ss.getSheets()[0].setName(DN.SHEET) : ss.insertSheet(DN.SHEET); }
  if (sh.getLastRow() === 0) {
    sh.appendRow(DN.HEAD);
    sh.getRange(1, 1, 1, DN.HEAD.length).setFontWeight('bold').setBackground('#0E2A2C').setFontColor('#FFFFFF');
    sh.setFrozenRows(1);
    sh.getRange('A:B').setNumberFormat('@');
    sh.getRange('H:H').setNumberFormat('@');
    sh.setColumnWidth(5, 520);
    sh.getRange('E:E').setWrap(true);
    sh.hideColumns(7, 2);
  }
  L.push('Таблица: ' + ss.getUrl());

  // папка голосовых
  let dir = null;
  if (dnGet_('VOICE_DIR')) { try { dir = DriveApp.getFolderById(dnGet_('VOICE_DIR')); } catch (e) { dir = null; } }
  if (!dir) { dir = DriveApp.createFolder(DN.VOICE_TITLE); dnSet_('VOICE_DIR', dir.getId()); }

  // доступ: только владелец
  [DriveApp.getFileById(ss.getId()), dir].forEach(function (f) {
    try { f.setSharing(DriveApp.Access.PRIVATE, DriveApp.Permission.NONE); } catch (e) {}
    const others = f.getEditors().concat(f.getViewers()).map(function (u) { return u.getEmail(); })
      .filter(function (m) { return m && m !== Session.getEffectiveUser().getEmail(); });
    L.push((others.length ? '⚠️ У «' + f.getName() + '» есть доступ у: ' + others.join(', ') : '✅ «' + f.getName() + '» – доступ только у Вас'));
  });

  // вебхук
  const url = dnGet_('WEBAPP_URL') || ScriptApp.getService().getUrl();
  if (!url) throw new Error('Сначала разверните веб-приложение: Развернуть → Новое развёртывание → Веб-приложение');
  if (!/\/exec$/.test(url)) throw new Error('Адрес веб-приложения должен оканчиваться на /exec. Впишите его в свойство WEBAPP_URL: ' + url);
  dnSet_('WEBAPP_URL', url);
  const me = tg_('getMe', {});
  if (!me.ok) throw new Error('Токен Telegram не работает: ' + tgErr_(me));
  const wh = tg_('setWebhook', { url: url + '?k=' + dnGet_('WH_KEY'), allowed_updates: ['message', 'callback_query'], drop_pending_updates: true });
  L.push(wh.ok ? '✅ Вебхук, бот @' + me.result.username : '❌ Вебхук не встал: ' + tgErr_(wh));
  tg_('setMyCommands', { commands: [{ command: 'start', description: 'Открыть дневник' }, { command: 'status', description: 'Проверка' }] });

  L.push(dnGet_('YC_KEY') && dnGet_('YC_FOLDER') ? '✅ Голосовые: SpeechKit' : '⚠️ Голосовые: нет YC_KEY или YC_FOLDER');
  if (dnGet_('OWNER_ID')) L.push(dnMenuButton_() ? '✅ Кнопка «Дневник» в чате бота' : '⚠️ Кнопка меню не встала');
  else L.push('', '>>> Отправьте боту @' + me.result.username + ' сообщение:  /start ' + dnGet_('SETUP_CODE'));
  L.push('Страница мини-приложения: ' + dnAppUrl_());
  console.log(L.join('\n'));
  return L.join('\n');
}

function dnStatus_() {
  const L = ['<b>Дневник ' + DN.VERSION + '</b>'];
  const wh = tg_('getWebhookInfo', {});
  L.push(wh.ok && wh.result.url ? '✅ Бот на связи' + (wh.result.last_error_message ? ' (последняя ошибка: ' + dnEsc_(wh.result.last_error_message) + ')' : '') : '❌ Вебхук не стоит');
  L.push(dnGet_('YC_KEY') ? '✅ Голосовые распознаются' : '⚠️ Голосовые не подключены');
  L.push(dnGet_('PIN_HASH') ? '✅ PIN-код задан' : '⚠️ PIN-код не задан – откройте дневник');
  L.push(dnGet_('CLEAN') === '1' ? '🧹 Ваши сообщения убираются из чата' : '• Сообщения в чате остаются');
  try { L.push('Записей сегодня: ' + dnTodayCount_()); } catch (e) { L.push('❌ Таблица: ' + dnEsc_(e.message)); }
  return L.join('\n');
}

function doGet() { return HtmlService.createHtmlOutput('Дневник ' + DN.VERSION); }

function doPost(e) {
  const q = (e && e.parameter) || {};
  if (q.k !== undefined) return dnWebhook_(e);
  return dnApp_(e);
}

function dnJson_(o) { return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON); }

function dnWebhook_(e) {
  const ok = HtmlService.createHtmlOutput('ok');     // HtmlService отвечает Telegram кодом 200 (ContentService – 302, Telegram повторяет)
  try {
    if (e.parameter.k !== dnGet_('WH_KEY')) return ok;
    const u = JSON.parse((e.postData && e.postData.contents) || '{}');
    if (!u.update_id) return ok;
    const cache = CacheService.getScriptCache();
    if (cache.get('u' + u.update_id)) return ok;
    cache.put('u' + u.update_id, '1', 3600);
    if (u.callback_query) dnCallback_(u.callback_query);
    else if (u.message) dnMessage_(u.message);
  } catch (err) { dnLog_('вебхук', err.message + '\n' + (err.stack || '')); }
  return ok;
}

/* ------------------------------ чат бота ------------------------------ */

function dnTagKb_(id, tag) {
  const b = function (t) { return { text: (tag === t ? '✅ ' : '') + DN.TAG_ICON[t] + ' ' + DN.TAGS[t], callback_data: 'tag:' + (tag === t ? 'none' : t) + ':' + id }; };
  return { inline_keyboard: [[b('idea'), b('personal')], [{ text: '↩️ Отменить запись', callback_data: 'undo:' + id }]] };
}
function dnTagFromText_(text) {
  let tag = '';
  if (/#(идея|идеи|idea)\b/i.test(text)) tag = 'idea';
  if (/#(личное|лично|personal)\b/i.test(text)) tag = 'personal';
  const clean = text.replace(/#(идея|идеи|idea|личное|лично|personal)(?=\s|$)/gi, '').replace(/[ \t]{2,}/g, ' ').trim();
  return { tag: tag, text: clean };
}
function dnSavedText_(r, tag, type, text) {
  let s = '✅ ' + r.time + ' · записал' + (type === 'голос' ? ' голосовое' : '') + (tag ? ' · ' + DN.TAG_ICON[tag] + ' ' + DN.TAGS[tag] : '');
  if (type === 'голос' && dnGet_('CLEAN') !== '1') s += '\n\n🎙 ' + dnEsc_(text || '(не разобрал слов)');
  return s;
}

function dnMessage_(m) {
  const from = m.from || {}, owner = dnGet_('OWNER_ID');
  if (!m.chat || m.chat.type !== 'private') return;
  const text = String(m.text || '').trim();

  if (!owner) {
    const code = String(dnGet_('SETUP_CODE')).toUpperCase();
    if (code && /^\/start/i.test(text) && text.toUpperCase().indexOf(code) >= 0) {
      dnSet_('OWNER_ID', String(from.id));
      dnMenuButton_();
      tg_('sendMessage', { chat_id: from.id, text: 'Готово, дневник привязан к Вам. Пишите или наговаривайте сюда в любой момент – всё попадёт в таблицу с датой и временем.\n\nКнопка «Дневник» слева от поля ввода – запись, метки и выгрузка с PIN-кодом.', reply_markup: dnAppKb_() });
    }
    return;
  }
  if (String(from.id) !== owner) return;                       // чужим – тишина

  if (/^\/start|^\/help/i.test(text)) {
    dnSay_('Пишите или наговаривайте сюда – всё попадёт в дневник.\nМетка: <b>#идея</b> или <b>#личное</b> в тексте или кнопкой под ответом.\n\nВыгрузка за день текстом или PDF – в кнопке «Дневник» (нужен PIN-код).', { reply_markup: dnAppKb_() });
    dnMenuButton_();
    return;
  }
  if (text === '/status') { dnSay_(dnStatus_()); return; }
  if (/^\//.test(text)) { dnSay_('Таких команд нет. Просто пишите – это запись в дневник. /status – проверка.'); return; }

  try {
    if (m.voice || (m.audio && /ogg|opus/i.test(m.audio.mime_type || ''))) { dnVoiceChat_(m); return; }
    if (!text) { dnSay_('В дневник пишу текст и голосовые.', { reply_to_message_id: m.message_id }); return; }
    const t = dnTagFromText_(text);
    const r = dnAdd_(t.text, t.tag, 'текст', '');
    dnAfterSave_(m, r, t.tag, 'текст', t.text);
  } catch (err) {
    dnLog_('запись', err.message);
    dnSay_('⚠️ Не записалось: ' + dnEsc_(err.message) + '. Повторите, пожалуйста.', { reply_to_message_id: m.message_id });
  }
}

function dnAfterSave_(m, r, tag, type, text) {
  const clean = dnGet_('CLEAN') === '1';
  if (clean) tg_('deleteMessage', { chat_id: m.chat.id, message_id: m.message_id });
  const p = { reply_markup: dnTagKb_(r.id, tag) };
  if (!clean) p.reply_to_message_id = m.message_id;
  dnSay_(dnSavedText_(r, tag, type, text), p);
}

function dnCallback_(q) {
  const ans = function (t) { tg_('answerCallbackQuery', { callback_query_id: q.id, text: t || '' }); };
  if (String((q.from || {}).id) !== dnGet_('OWNER_ID')) { ans(); return; }
  const parts = String(q.data || '').split(':'), chat = q.message.chat.id, mid = q.message.message_id;
  try {
    if (parts[0] === 'tag') {
      const tag = parts[1] === 'none' ? '' : parts[1], id = parts[2];
      if (!dnSetTag_(id, tag)) { ans('Запись не найдена'); return; }
      const txt = String(q.message.text || '').replace(/ · (💡 Идея|🔒 Личное)/, '').replace(/(записал( голосовое)?)/, '$1' + (tag ? ' · ' + DN.TAG_ICON[tag] + ' ' + DN.TAGS[tag] : ''));
      tg_('editMessageText', { chat_id: chat, message_id: mid, text: txt, reply_markup: dnTagKb_(id, tag) });
      ans(tag ? 'Метка: ' + DN.TAGS[tag] : 'Метка снята');
    } else if (parts[0] === 'undo') {
      const err = dnUndo_(parts[1]);
      if (err) { ans(err); return; }
      tg_('editMessageText', { chat_id: chat, message_id: mid, text: '↩️ Запись отменена' });
      ans('Отменено');
    } else ans();
  } catch (err) { ans('Ошибка: ' + err.message); }
}

function dnVoiceDir_() { return DriveApp.getFolderById(dnGet_('VOICE_DIR')); }

function dnVoiceChat_(m) {
  const v = m.voice || m.audio;
  const name = dnFmt_(new Date(), 'yyyy-MM-dd_HHmmss') + '.ogg';
  const blob = dnFileBlob_(v.file_id, name);
  const audio = dnVoiceDir_().createFile(blob).getId();
  const t = dnTagFromText_(String(m.caption || ''));
  if (!dnGet_('YC_KEY') || !dnGet_('YC_FOLDER')) {
    const r = dnAdd_('', t.tag, 'голос', audio);
    dnAfterSave_(m, r, t.tag, 'голос', '(распознавание не подключено – сохранил файл)');
    return;
  }
  const bytes = blob.getBytes();
  if ((v.duration || 0) <= DN.STT_SYNC_SEC && bytes.length <= DN.STT_SYNC_BYTES) {
    try {
      const text = dnSttOgg_(bytes);
      const r = dnAdd_(text, t.tag, 'голос', audio);
      dnAfterSave_(m, r, t.tag, 'голос', text);
      return;
    } catch (err) { dnLog_('распознавание', err.message); }
  }
  const op = dnSttAsyncStart_(bytes);
  dnSet_('VOP_' + op, JSON.stringify({ audio: audio, tag: t.tag, chat: m.chat.id, msg: m.message_id, t: Date.now() }));
  dnPollLater_();
  dnSay_('🎙 Длинное голосовое – распознаю, запишу в течение пары минут.', { reply_to_message_id: m.message_id });
}

/** Запустить dnPoll через минуту (только когда есть длинные голосовые – без постоянного триггера). */
function dnPollLater_() {
  const has = ScriptApp.getProjectTriggers().some(function (t) { return t.getHandlerFunction() === 'dnPoll'; });
  if (!has) ScriptApp.newTrigger('dnPoll').timeBased().after(60 * 1000).create();
}

function dnPoll() {
  ScriptApp.getProjectTriggers().forEach(function (t) { if (t.getHandlerFunction() === 'dnPoll') ScriptApp.deleteTrigger(t); });
  const all = dnP_().getProperties();
  const h = { Authorization: 'Api-Key ' + dnGet_('YC_KEY'), 'x-folder-id': dnGet_('YC_FOLDER') };
  let left = 0;
  Object.keys(all).filter(function (k) { return k.indexOf('VOP_') === 0; }).forEach(function (k) {
    const op = k.slice(4), job = JSON.parse(all[k]);
    const m = { chat: { id: job.chat }, message_id: job.msg };
    try {
      const st = JSON.parse(UrlFetchApp.fetch('https://operation.api.cloud.yandex.net/operations/' + op, { headers: h, muteHttpExceptions: true }).getContentText() || '{}');
      if (!st.done) {
        if (Date.now() - job.t < 20 * 60000) { left++; return; }
        throw new Error('распознавание не закончилось за 20 минут');
      }
      if (st.error) throw new Error(st.error.message || 'ошибка распознавания');
      const raw = UrlFetchApp.fetch('https://stt.api.cloud.yandex.net/stt/v3/getRecognition?operationId=' + op, { headers: h, muteHttpExceptions: true }).getContentText();
      const text = dnSttV3Text_(raw);
      const r = dnAdd_(text, job.tag, 'голос', job.audio);
      dnAfterSave_(m, r, job.tag, 'голос', text);
    } catch (err) {
      const r = dnAdd_('', job.tag, 'голос', job.audio);
      dnSay_('⚠️ Голосовое сохранил файлом, текст не распознал: ' + dnEsc_(err.message), { reply_to_message_id: job.msg, reply_markup: dnTagKb_(r.id, job.tag) });
    }
    dnDel_(k);
  });
  if (left) dnPollLater_();
}

/* ============================== SpeechKit ============================== */

function dnSttOgg_(bytes) {
  const url = 'https://stt.api.cloud.yandex.net/speech/v1/stt:recognize?lang=ru-RU&format=oggopus&folderId=' + encodeURIComponent(dnGet_('YC_FOLDER'));
  const r = UrlFetchApp.fetch(url, { method: 'post', contentType: 'application/octet-stream', payload: bytes, headers: { Authorization: 'Api-Key ' + dnGet_('YC_KEY') }, muteHttpExceptions: true });
  const j = JSON.parse(r.getContentText() || '{}');
  if (r.getResponseCode() !== 200) throw new Error('SpeechKit: ' + (j.error_message || r.getResponseCode()));
  return String(j.result || '').trim();
}

/** Куски PCM 16 кГц из мини-приложения (каждый до 25 с) – распознаём параллельно. */
function dnSttPcm_(chunks) {
  const url = 'https://stt.api.cloud.yandex.net/speech/v1/stt:recognize?lang=ru-RU&format=lpcm&sampleRateHertz=16000&folderId=' + encodeURIComponent(dnGet_('YC_FOLDER'));
  const reqs = chunks.map(function (b64) {
    return { url: url, method: 'post', contentType: 'application/octet-stream', payload: Utilities.base64Decode(b64), headers: { Authorization: 'Api-Key ' + dnGet_('YC_KEY') }, muteHttpExceptions: true };
  });
  const out = [];
  for (let i = 0; i < reqs.length; i += 8) {
    UrlFetchApp.fetchAll(reqs.slice(i, i + 8)).forEach(function (r) {
      const j = JSON.parse(r.getContentText() || '{}');
      if (r.getResponseCode() !== 200) throw new Error('SpeechKit: ' + (j.error_message || r.getResponseCode()));
      if (j.result) out.push(String(j.result).trim());
    });
  }
  return out.join(' ').replace(/\s+/g, ' ').trim();
}

function dnSttAsyncStart_(bytes) {
  const body = {
    content: Utilities.base64Encode(bytes),
    recognitionModel: {
      model: 'general',
      audioFormat: { containerAudio: { containerAudioType: 'OGG_OPUS' } },
      textNormalization: { textNormalization: 'TEXT_NORMALIZATION_ENABLED', literatureText: true },
      languageRestriction: { restrictionType: 'WHITELIST', languageCode: ['ru-RU'] }
    }
  };
  const r = UrlFetchApp.fetch('https://stt.api.cloud.yandex.net/stt/v3/recognizeFileAsync', {
    method: 'post', contentType: 'application/json', payload: JSON.stringify(body), muteHttpExceptions: true,
    headers: { Authorization: 'Api-Key ' + dnGet_('YC_KEY'), 'x-folder-id': dnGet_('YC_FOLDER') }
  });
  const j = JSON.parse(r.getContentText() || '{}');
  if (!j.id) throw new Error('SpeechKit не принял голосовое: ' + (j.message || r.getResponseCode()));
  return j.id;
}

function dnSttV3Text_(raw) {
  const fin = [], ref = [];
  String(raw || '').split('\n').forEach(function (line) {
    if (!line.trim()) return;
    let o; try { o = JSON.parse(line); } catch (e) { return; }
    const res = o.result || o;
    const a = res.finalRefinement && res.finalRefinement.normalizedText && res.finalRefinement.normalizedText.alternatives;
    const b = res.final && res.final.alternatives;
    if (a && a[0] && a[0].text) ref.push(a[0].text);
    else if (b && b[0] && b[0].text) fin.push(b[0].text);
  });
  return (ref.length ? ref : fin).join(' ').trim();
}

/** Проверка подписи Telegram (initData) и что это владелец. Возвращает user. */
function dnAuth_(init) {
  if (!init) throw new Error('Откройте дневник кнопкой «Дневник» в чате бота');
  const m = {};
  String(init).split('&').forEach(function (s) {
    const i = s.indexOf('=');
    if (i > 0) m[s.slice(0, i)] = decodeURIComponent(s.slice(i + 1).replace(/\+/g, '%20'));
  });
  const hash = m.hash;
  delete m.hash;
  const dcs = Object.keys(m).sort().map(function (k) { return k + '=' + m[k]; }).join('\n');
  const secret = Utilities.computeHmacSha256Signature(Utilities.newBlob(dnGet_('TG_TOKEN')).getBytes(), Utilities.newBlob('WebAppData').getBytes());
  const sig = dnHex_(Utilities.computeHmacSha256Signature(Utilities.newBlob(dcs).getBytes(), secret));
  let user = {};
  try { user = JSON.parse(m.user || '{}'); } catch (e) {}
  if (!hash || sig !== hash || String(user.id) !== dnGet_('OWNER_ID')) {
    const c = CacheService.getScriptCache();
    if (!c.get('alien')) {
      c.put('alien', '1', 3600);
      dnSay_('⚠️ Кто-то пытался открыть дневник не с Вашего аккаунта' + (user.id && String(user.id) !== dnGet_('OWNER_ID') ? ' (Telegram ID ' + dnEsc_(user.id) + (user.username ? ', @' + dnEsc_(user.username) : '') + ')' : '') + '. Доступа он не получил.');
    }
    throw new Error('Нет доступа');
  }
  if (Date.now() / 1000 - Number(m.auth_date || 0) > DN.AUTH_MAX_SEC) throw new Error('Сессия устарела – закройте дневник и откройте заново');
  return user;
}

function dnApp_(e) {
  let body = {};
  try { body = JSON.parse((e.postData && e.postData.contents) || '{}'); } catch (x) { return dnJson_({ ok: false, error: 'Неверный запрос' }); }
  try {
    dnAuth_(body.init);
    const op = body.op;
    if (op === 'hello') return dnJson_({ ok: true, version: DN.VERSION, pin: !!dnGet_('PIN_HASH'), today: dnTodayCount_(), clean: dnGet_('CLEAN') === '1', stt: !!dnGet_('YC_KEY'), date: dnFmt_(new Date(), 'yyyy-MM-dd') });
    if (op === 'save') {
      const tag = DN.TAGS[body.tag] ? body.tag : '';
      const r = dnAdd_(String(body.text || '').slice(0, 20000), tag, body.audio ? 'голос' : 'текст', body.audio ? String(body.audio) : '');
      return dnJson_({ ok: true, id: r.id, time: r.time, today: dnTodayCount_() });
    }
    if (op === 'stt') return dnJson_(dnAppStt_(body));
    if (op === 'undo') { const err = dnUndo_(body.id); return dnJson_(err ? { ok: false, error: err } : { ok: true, today: dnTodayCount_() }); }
    if (op === 'pinset') return dnJson_(dnPinSet_(body));
    if (op === 'clean') { dnPinCheck_(body.pin); dnSet_('CLEAN', body.on ? '1' : '0'); return dnJson_({ ok: true, clean: !!body.on }); }
    if (op === 'export') return dnJson_(dnExport_(body));
    return dnJson_({ ok: false, error: 'Неизвестная операция' });
  } catch (err) {
    return dnJson_({ ok: false, error: err.message, lock: err.lock || 0 });
  }
}

/** Голос из приложения: исходный файл → папка голосовых, PCM-куски → текст. Запись не создаёт – текст возвращается в поле. */
function dnAppStt_(b) {
  if (!dnGet_('YC_KEY') || !dnGet_('YC_FOLDER')) throw new Error('Распознавание не подключено (нет YC_KEY)');
  const chunks = (b.pcm || []).slice(0, 16);                    // до 16 кусков по 25 с ≈ 6 минут
  if (!chunks.length) throw new Error('Пустая запись');
  let audio = '';
  if (b.audio) {
    const ext = /mp4|aac|m4a/i.test(b.mime || '') ? 'm4a' : (/ogg/i.test(b.mime || '') ? 'ogg' : 'webm');
    const blob = Utilities.newBlob(Utilities.base64Decode(b.audio), b.mime || 'audio/webm', dnFmt_(new Date(), 'yyyy-MM-dd_HHmmss') + '.' + ext);
    audio = dnVoiceDir_().createFile(blob).getId();
  }
  return { ok: true, text: dnSttPcm_(chunks), audio: audio };
}

/* ------------------------------ PIN ------------------------------ */

function dnPinHash_(pin) { return dnSha_(dnGet_('PIN_SALT') + '|' + String(pin)); }

function dnPinCheck_(pin) {
  if (!dnGet_('PIN_HASH')) { const e = new Error('PIN-код ещё не задан'); throw e; }
  const lock = Number(dnGet_('PIN_LOCK')) || 0;
  if (lock > Date.now()) { const e = new Error('Выгрузка заблокирована после ' + DN.PIN_TRIES + ' неверных PIN-кодов. Попробуйте после ' + dnFmt_(new Date(lock), 'HH:mm')); e.lock = lock; throw e; }
  if (/^\d{4,6}$/.test(String(pin || '')) && dnPinHash_(pin) === dnGet_('PIN_HASH')) { dnDel_('PIN_FAIL'); return true; }
  const n = (Number(dnGet_('PIN_FAIL')) || 0) + 1;
  if (n >= DN.PIN_TRIES) {
    const until = Date.now() + DN.PIN_LOCK_MIN * 60000;
    dnSet_('PIN_LOCK', until);
    dnDel_('PIN_FAIL');
    dnSay_('⚠️ В дневнике ' + DN.PIN_TRIES + ' раз подряд ввели неверный PIN-код. Выгрузка закрыта до ' + dnFmt_(new Date(until), 'HH:mm') + '.');
    const e = new Error('Неверный PIN-код. Выгрузка заблокирована на ' + DN.PIN_LOCK_MIN + ' минут'); e.lock = until; throw e;
  }
  dnSet_('PIN_FAIL', n);
  throw new Error('Неверный PIN-код (осталось попыток: ' + (DN.PIN_TRIES - n) + ')');
}

function dnPinSet_(b) {
  const pin = String(b.pin || '');
  if (!/^\d{4,6}$/.test(pin)) throw new Error('PIN-код – от 4 до 6 цифр');
  if (dnGet_('PIN_HASH')) dnPinCheck_(b.old);                    // смена – только со старым кодом
  dnSet_('PIN_SALT', dnRand_(24));
  dnSet_('PIN_HASH', dnPinHash_(pin));
  dnDel_('PIN_FAIL');
  dnSay_('🔐 PIN-код дневника ' + (b.old ? 'изменён' : 'задан') + ' (' + dnFmt_(new Date(), 'dd.MM HH:mm') + ').');
  return { ok: true };
}

/* ------------------------------ выгрузка ------------------------------ */

function dnRows_(from, to, tag) {
  const sh = dnSheet_(), last = sh.getLastRow();
  if (last < 2) return [];
  const want = tag && DN.TAGS[tag] ? DN.TAGS[tag] : '';
  return sh.getRange(2, 1, last - 1, 8).getDisplayValues()
    .filter(function (r) { return r[7] >= from && r[7] <= to && (!want || r[2] === want) && (r[4] || r[5]); })
    .map(function (r) { return { day: r[7], time: r[1], tag: r[2], type: r[3], text: String(r[4]).replace(/^'/, ''), audio: r[5] }; })
    .sort(function (a, b) { return a.day === b.day ? (a.time < b.time ? -1 : 1) : (a.day < b.day ? -1 : 1); });
}

function dnTagKey_(name) { return name === DN.TAGS.idea ? 'idea' : (name === DN.TAGS.personal ? 'personal' : ''); }

function dnPeriodTitle_(from, to) { return from === to ? dnDot_(from) : dnDot_(from) + ' – ' + dnDot_(to); }
function dnFilterTitle_(tag) { return tag === 'idea' ? 'идеи' : (tag === 'personal' ? 'личное' : 'все записи'); }

function dnPlain_(rows, from, to, tag) {
  const L = ['Дневник · ' + dnPeriodTitle_(from, to) + ' · ' + dnFilterTitle_(tag)];
  let day = '';
  rows.forEach(function (r) {
    if (r.day !== day) { day = r.day; L.push('', '— ' + dnDayTitle_(day) + ' —'); }
    const k = dnTagKey_(r.tag);
    L.push('', r.time + (r.type === 'голос' ? ' · 🎙' : '') + (k ? ' · ' + DN.TAG_ICON[k] + ' ' + r.tag : ''));
    L.push(r.text || '(голосовое без текста)');
  });
  return L.join('\n');
}

function dnPdf_(rows, from, to, tag) {
  const css = 'body{font-family:Arial,sans-serif;color:#10282A;font-size:11pt;line-height:1.45;margin:0}' +
    '.top{background:#0E2A2C;color:#fff;padding:18px 22px;border-radius:8px}' +
    '.top h1{margin:0;font-size:20pt;letter-spacing:.5px}.top p{margin:4px 0 0;color:#9FE8E0;font-size:10pt}' +
    'h2{font-size:12pt;color:#0E7C74;margin:22px 0 6px;padding-bottom:4px;border-bottom:1px solid #CFE9E6;text-transform:capitalize}' +
    '.e{margin:0 0 12px;padding:8px 12px;border-left:3px solid #19C9B6;background:#F4FBFA}' +
    '.m{font-size:9pt;color:#5B7C7A;margin-bottom:3px}.t{white-space:pre-wrap}' +
    '.tag{display:inline-block;padding:0 6px;border-radius:8px;font-size:8.5pt;margin-left:6px}' +
    '.idea{background:#FFF2C6;color:#7A5A00}.personal{background:#E6E2FF;color:#3B2F8F}' +
    '.f{margin-top:26px;font-size:8.5pt;color:#8AA5A3;text-align:center}';
  let h = '<html><head><meta charset="utf-8"><style>' + css + '</style></head><body>';
  h += '<div class="top"><h1>Дневник</h1><p>' + dnEsc_(dnPeriodTitle_(from, to)) + ' · ' + dnEsc_(dnFilterTitle_(tag)) + ' · записей: ' + rows.length + '</p></div>';
  let day = '';
  rows.forEach(function (r) {
    if (r.day !== day) { day = r.day; h += '<h2>' + dnEsc_(dnDayTitle_(day)) + '</h2>'; }
    const k = dnTagKey_(r.tag);
    h += '<div class="e"><div class="m">' + dnEsc_(r.time) + (r.type === 'голос' ? ' · голосовое' : '') +
      (k ? '<span class="tag ' + k + '">' + dnEsc_(r.tag) + '</span>' : '') + '</div>' +
      '<div class="t">' + dnEsc_(r.text || '(голосовое без текста)') + '</div></div>';
  });
  h += '<div class="f">Выгружено ' + dnFmt_(new Date(), 'dd.MM.yyyy HH:mm') + '</div></body></html>';
  const name = 'Дневник ' + dnPeriodTitle_(from, to).replace(/ – /, '–') + (tag ? ' (' + dnFilterTitle_(tag) + ')' : '') + '.pdf';
  return Utilities.newBlob(h, 'text/html', 'd.html').getAs('application/pdf').setName(name);
}

function dnExport_(b) {
  dnPinCheck_(b.pin);
  const from = String(b.from || ''), to = String(b.to || b.from || '');
  if (!/^\d{4}-\d{2}-\d{2}$/.test(from) || !/^\d{4}-\d{2}-\d{2}$/.test(to)) throw new Error('Выберите дату');
  if (to < from) throw new Error('Конец периода раньше начала');
  const span = (new Date(to + 'T12:00:00Z') - new Date(from + 'T12:00:00Z')) / 86400000;
  if (span > DN.MAX_DAYS) throw new Error('Период не длиннее ' + DN.MAX_DAYS + ' дней');
  const tag = DN.TAGS[b.tag] ? b.tag : '';
  const rows = dnRows_(from, to, tag);
  if (!rows.length) return { ok: true, count: 0, text: '' };

  if (b.fmt === 'pdf') {
    const r = tg_('sendDocument', { chat_id: dnGet_('OWNER_ID'), document: dnPdf_(rows, from, to, tag), caption: '📄 Дневник · ' + dnPeriodTitle_(from, to) + ' · ' + dnFilterTitle_(tag) + ' · ' + rows.length + ' зап.' }, true);
    if (!r.ok) throw new Error('PDF не отправился: ' + tgErr_(r));
    return { ok: true, count: rows.length, sent: 'pdf' };
  }
  const text = dnPlain_(rows, from, to, tag);
  if (b.chat) {
    dnSplit_(text, DN.TG_TEXT).forEach(function (part) {
      const r = tg_('sendMessage', { chat_id: dnGet_('OWNER_ID'), text: part, disable_web_page_preview: true });
      if (!r.ok) throw new Error('В чат не отправилось: ' + tgErr_(r));
    });
  }
  return { ok: true, count: rows.length, text: text, sent: b.chat ? 'text' : '' };
}
